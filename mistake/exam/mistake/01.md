```C
#include <stdio.h>

int main(void)
{
    unsigned int x = 0x12345678;
    unsigned char *p = (unsigned char *)&x;

    printf("%02x %02x %02x %02x\n",
           p[0], p[1], p[2], p[3]);

    unsigned int y = ((unsigned int)p[0] << 24) |
                     ((unsigned int)p[1] << 16) |
                     ((unsigned int)p[2] << 8)  |
                     p[3];

    printf("%08x\n", y);

    return 0;
}
```
